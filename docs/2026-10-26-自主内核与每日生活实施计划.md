import json, os
from pathlib import Path

def atomic_write_json(path: Path, data: object) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(path.suffix + ".tmp")
    tmp.write_text(json.dumps(data, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    os.replace(tmp, path)

def read_json(path: Path, default: object) -> object:
    if not path.exists():
        return default
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError):
        return default

def append_jsonl(path: Path, row: dict) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("a", encoding="utf-8") as f:
        f.write(json.dumps(row, ensure_ascii=False) + "\n")
```

- [ ] **Step 3: 实现 tick 规则**

`tick` 按顺序检查：暂停文件、静默时段、每日活动次数、120 分钟间隔、下一个目标、权限。边界如下：

- 静默时段：`00:00 <= now.time() < 07:30`。
- 每日活动日志文件名：`activities/<YYYY-MM-DD>.jsonl`。
- 每日最多 4 条 `status == "done"` 记录。
- 最近活动时间与 now 的差值小于 120 分钟则返回 `activity_gap`。
- 权限不是 `allow` 时返回 `permission_blocked`，不得调用 runner。
- runner 成功：目标 `done`，写活动日志和审计。
- runner 失败：目标 `failed`，写失败结果和审计；下一次 tick 可选择该目标重试，但同样受间隔限制。

import argparse, sys, time
from datetime import datetime
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parents[1]))
from xiaojuju_core.daemon import tick
from xiaojuju_core.paths import Paths

def main():
    ap=argparse.ArgumentParser(); ap.add_argument("--once",action="store_true"); ap.add_argument("--loop",action="store_true"); ap.add_argument("--pause",action="store_true"); ap.add_argument("--resume",action="store_true"); ap.add_argument("--status",action="store_true"); ap.add_argument("--dry-run",action="store_true")
    a=ap.parse_args(); p=Paths.production()
    if a.pause: p.paused.parent.mkdir(parents=True,exist_ok=True); p.paused.write_text("paused",encoding="utf-8"); return
    if a.resume: p.paused.unlink(missing_ok=True); return
    if a.status: print("paused" if p.paused.exists() else "running"); return
    if a.once: print(tick(p, datetime.now(), dry_run=a.dry_run)); return
    while True:
        print(tick(p, datetime.now(), dry_run=a.dry_run)); time.sleep(900)

if __name__ == "__main__": main()
```
